# Phase 4: Religious State
# The active model follows the Church and State law. An atheist state religion
# is also treated as secular so console-driven or legacy transitions stay safe.

ve_is_atheist_state = {
	OR = {
		has_law = law_type:law_state_atheism
		country_has_state_religion = rel:atheist
	}
}

ve_is_confessional_state = {
	has_law = law_type:law_state_religion
	NOT = {
		ve_is_atheist_state = yes
	}
}

ve_is_pluralist_state = {
	NOR = {
		ve_is_atheist_state = yes
		ve_is_confessional_state = yes
	}
}

ve_missionaries_enabled = {
	OR = {
		ve_is_confessional_state = yes
		has_law = law_type:law_freedom_of_conscience
	}
	NOT = {
		ve_is_atheist_state = yes
	}
}

ve_religious_program_affordable = {
	ve_religious_state_value >= 50
	NOT = { has_modifier = ve_religious_program_strain_mdf }
}

ve_has_religious_program_1 = {
	OR = {
		has_modifier = ve_confessional_program_1_mdf
		has_modifier = ve_pluralist_program_1_mdf
		has_modifier = ve_atheist_program_1_mdf
