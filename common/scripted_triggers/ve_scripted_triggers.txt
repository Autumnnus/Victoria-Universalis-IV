age_1_obj_triggers = {
	AND = {
		root.average_sol >= 10
		AND = {
			has_law = law_type:law_slavery_banned
			NOT = {
				has_law = law_type:law_serfdom
			}
		}
		government_legitimacy >= 75
		taking_loans = no
		ig:ig_intelligentsia = {
			is_in_government = yes
			ig_clout >= 0.15
		}
		pop_type_percent_country = {
			pop_type = peasants
			percent < 0.5
		}
		num_declared_interests >= 8
	}
}

age_2_obj_triggers = {
	AND = {
		OR = {
			has_law = law_type:law_colonial_resettlement
			has_law = law_type:law_colonial_exploitation
		}
		any_subject_or_below = {
			OR = {
				has_diplomatic_pact = {
					who = ROOT
					type = puppet
				}
				has_diplomatic_pact = {
					who = ROOT
					type = dominion
				}
			}
			count >= 2
		}
		any_scope_state = {
			owner = ROOT
			state_population > 250000
			has_building = building_railway
		}
		navy_size >= 120
		leading_producer_of = g:coal
		literacy_rate >= 0.5
		total_population > 100000000
	}
}

# Faz 5c - can this country afford the NEXT idea of $GROUP$?
# var:$GROUP$_idea holds how many are already bought, so level N+1 costs IDEA_COST[N+1].
ve_idea_affordable = {
	OR = {
		AND = {
			var:$GROUP$_idea == 0
			idea_point_value >= 200
		}
		AND = {
			var:$GROUP$_idea == 1
			idea_point_value >= 250
		}
		AND = {
			var:$GROUP$_idea == 2
			idea_point_value >= 300
		}
		AND = {
			var:$GROUP$_idea == 3
			idea_point_value >= 375
		}
		AND = {
			var:$GROUP$_idea == 4
			idea_point_value >= 450
		}
		AND = {
			var:$GROUP$_idea == 5
			idea_point_value >= 550
		}
		AND = {
			var:$GROUP$_idea == 6
			idea_point_value >= 675
		}
	}
}
