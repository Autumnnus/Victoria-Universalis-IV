# Phase 6 Era Crossroads JE shells.
# Each JE is reachable only through its completed Age-specific offer pack.

ve_je_era_crossroads_age_1 = {
	icon = "gfx/interface/icons/building_icons/building_railway.dds"
	group = je_group_internal_affairs
	on_monthly_pulse = { effect = { ve_tick_era_crossroads = yes } }
	complete = { var:ve_era_crossroads_progress >= 60 }
	on_complete = { ve_complete_era_crossroads = yes }
	invalid = { NOT = { ve_has_valid_era_crossroads_context = yes } }
	on_invalid = { ve_invalidate_era_crossroads = yes }
	timeout = 1825
	on_timeout = { ve_timeout_era_crossroads = yes }
	current_value = { value = root.var:ve_era_crossroads_progress }
	goal_add_value = { add = 60 }
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = ve_je_era_crossroads_age_1_route_1_status
				trigger = { var:ve_era_crossroads_route = 1 }
			}
			triggered_desc = {
				desc = ve_je_era_crossroads_age_1_route_2_status
				trigger = { var:ve_era_crossroads_route = 2 }
			}
			triggered_desc = {
				desc = ve_je_era_crossroads_age_1_route_3_status
				trigger = { var:ve_era_crossroads_route = 3 }
			}
			triggered_desc = {
				desc = ve_je_era_crossroads_age_1_status
				trigger = { always = yes }
			}
		}
	}
	progressbar = yes
	weight = 100
	transferable = no
	can_revolution_inherit = no
	should_be_pinned_by_default_uninvolved_or_context = yes
}
