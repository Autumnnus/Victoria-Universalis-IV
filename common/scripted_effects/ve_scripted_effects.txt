# Every purchased group idea contributes one national-idea progress point.
# National ideas unlock at 3/6/9/.../21 total points. The cap keeps the
# national profile at its seven intended ideas even if more groups are opened.
# Recalculation also makes group cancellation deterministic.
ve_recalculate_national_idea_pool = {
	# Nationalist ideas 5 and 7 use variables for their mechanical effects.
	# Mirror existing saves into permanent ownership modifiers so every group
	# icon can consistently use its modifier as the active-state source.
	if = {
		limit = {
			has_variable = nationalist_idea_5_var
			NOT = { has_modifier = nationalist_idea_5_mdf }
		}
		add_modifier = { name = nationalist_idea_5_mdf months = -1 }
	}
	if = {
		limit = {
			has_variable = nationalist_idea_7_var
			NOT = { has_modifier = nationalist_idea_7_mdf }
		}
		add_modifier = { name = nationalist_idea_7_mdf months = -1 }
	}
	set_variable = {
		name = national_idea_pool
		value = 0
	}
	change_variable = { name = national_idea_pool add = var:academic_idea }
	change_variable = { name = national_idea_pool add = var:agrarian_idea }
	change_variable = { name = national_idea_pool add = var:financial_idea }
	change_variable = { name = national_idea_pool add = var:infrastructure_idea }
	change_variable = { name = national_idea_pool add = var:extraction_idea }
	change_variable = { name = national_idea_pool add = var:mass_conscription_idea }
	change_variable = { name = national_idea_pool add = var:war_economy_idea }
	change_variable = { name = national_idea_pool add = var:industrialist_idea }
	change_variable = { name = national_idea_pool add = var:bureaucratic_idea }
	change_variable = { name = national_idea_pool add = var:nationalist_idea }
	change_variable = { name = national_idea_pool add = var:parliamentary_idea }
	change_variable = { name = national_idea_pool add = var:diplomatic_idea }
	change_variable = { name = national_idea_pool add = var:mercantile_idea }
	change_variable = { name = national_idea_pool add = var:welfare_idea }
	change_variable = { name = national_idea_pool add = var:reformist_idea }
	change_variable = { name = national_idea_pool add = var:labour_idea }
	change_variable = { name = national_idea_pool add = var:offensive_idea }
	change_variable = { name = national_idea_pool add = var:defensive_idea }
	change_variable = { name = national_idea_pool add = var:naval_idea }
	change_variable = { name = national_idea_pool add = var:professional_army_idea }
	change_variable = { name = national_idea_pool add = var:colonial_idea }
	if = {
		limit = { var:national_idea_pool > 21 }
		set_variable = { name = national_idea_pool value = 21 }
	}
}
