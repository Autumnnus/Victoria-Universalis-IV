# Records the first time each objective is completed in the current age.
# Losing an objective stops its monthly income, but never grants the +50 again.
ve_initialize_era_momentum = {
	if = {
		limit = { NOT = { has_variable = ve_splender_points } }
		set_variable = { name = ve_splender_points value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = ve_triggered_objectives } }
		set_variable = { name = ve_triggered_objectives value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = ve_age_rewards_purchased } }
		set_variable = {
			name = ve_age_rewards_purchased
			value = ve_current_age_rewards_owned
		}
	}
	if = {
		limit = {
			has_variable = clicked_golden_age
			NOT = { has_variable = ve_golden_age_used }
		}
		set_variable = ve_golden_age_used
	}
	if = {
		limit = { var:ve_splender_points < 0 }
		set_variable = { name = ve_splender_points value = 0 }
	}
}

ve_purchase_era_reward = {
	if = {
		limit = { var:ve_age_rewards_purchased <= 0 }
		change_variable = { name = ve_splender_points subtract = 200 }
	}
	else_if = {
		limit = { var:ve_age_rewards_purchased == 1 }
		change_variable = { name = ve_splender_points subtract = 300 }
	}
	else_if = {
		limit = { var:ve_age_rewards_purchased == 2 }
		change_variable = { name = ve_splender_points subtract = 400 }
	}
	else_if = {
		limit = { var:ve_age_rewards_purchased == 3 }
		change_variable = { name = ve_splender_points subtract = 500 }
	}
	else = {
		change_variable = { name = ve_splender_points subtract = 600 }
	}
	change_variable = {
		name = ve_age_rewards_purchased
		add = 1
	}
}

# AI deliberately takes the first affordable untaken reward. This keeps AI
# spending deterministic and uses the same cost curve as the player.
ve_ai_uses_age_bonuses = {
	if = {
		limit = {
			ve_can_purchase_era_reward = yes
			OR = {
				AND = { has_global_variable = ve_age_1_started NOT = { has_modifier = age_1_bonus_1 } }
				AND = { has_global_variable = ve_age_2_started NOT = { has_modifier = age_2_bonus_1 } }
				AND = { has_global_variable = ve_age_3_started NOT = { has_modifier = age_3_bonus_1 } }

		}
