set_national_idea = {
	# Each tag is guarded with exists: SGF, NGF and PRU are not present when this runs during
	# setup, and comparing against a country that does not exist raises
	# "Invalid right side during comparison 'c'". Same pattern the rest of this file uses.
	if = {
		limit = {
			NOR = {
				AND = {
					exists = c:GBR
					this = c:GBR
				}
				AND = {
					exists = c:FRA
					this = c:FRA
				}
				AND = {
					exists = c:GER
					this = c:GER
				}
				AND = {
					exists = c:SGF
					this = c:SGF
				}
				AND = {
					exists = c:NGF
					this = c:NGF
				}
				AND = {
					exists = c:PRU
					this = c:PRU
				}
				AND = {
					exists = c:AUS
					this = c:AUS
				}
				AND = {
					exists = c:RUS
					this = c:RUS
				}
				AND = {
					exists = c:TUR
					this = c:TUR
				}
				AND = {
					exists = c:CHI
					this = c:CHI
				}
				AND = {
					exists = c:JAP
					this = c:JAP
				}
				AND = {
					exists = c:USA
					this = c:USA
				}
			}
		}
		set_variable = generic_idea_country
	}
}
