# Scope: country. Called only after the core begin effect has set progress to
# zero and, for targeted projects, stored the target state.
ve_evt_begin_religious_incident_session = {
	ve_evt_clear_religious_incident_session = yes
	ve_evt_clear_religious_je_resolution = yes
	set_variable = {
		name = ve_evt_religious_incident_count
		value = 0
	}
	add_modifier = {
		name = ve_evt_religious_incident_cooldown_mdf
		months = 3
	}
	if = {
		limit = {
			NOT = { has_journal_entry = ve_je_religious_project }
			has_variable = ve_religious_project_target_state
		}
		# Journal Entry targets require a named scope. Passing the state variable
		# directly leaves SCOPE.sState('target') unset in the JE localization.
		var:ve_religious_project_target_state ?= {
			save_temporary_scope_as = ve_religious_project_je_target
		}
		add_journal_entry = {
			type = ve_je_religious_project
			target = scope:ve_religious_project_je_target
		}
	}
}

# Countrywide Cultural Projects do not have a state target, so they receive the
# same JE without a target. Targeted projects pass their stored state.
ve_evt_begin_cultural_incident_session = {
	ve_evt_clear_cultural_incident_session = yes
	ve_evt_clear_cultural_je_resolution = yes
	set_variable = {
		name = ve_evt_cultural_incident_count
		value = 0
	}
	add_modifier = {
		name = ve_evt_cultural_incident_cooldown_mdf
		months = 3
	}
	if = {
		limit = {
			NOT = { has_journal_entry = ve_je_cultural_project }
			has_variable = ve_cultural_project_target_state
		}
		# See the religious-project equivalent above: the JE target must receive
		# an explicit saved scope, not a variable expression.
		var:ve_cultural_project_target_state ?= {
			save_temporary_scope_as = ve_cultural_project_je_target
		}
		add_journal_entry = {
			type = ve_je_cultural_project
			target = scope:ve_cultural_project_je_target
		}
	}
	else_if = {
		limit = {
			NOT = { has_journal_entry = ve_je_cultural_project }
		}
		add_journal_entry = {
			type = ve_je_cultural_project
		}
	}
}

ve_evt_change_religious_project_progress = {
	if = {
		limit = { ve_has_active_religious_project = yes }
		change_variable = {
			name = ve_religious_project_progress
			add = $DELTA$
		}
		if = {
			limit = { var:ve_religious_project_progress < 0 }
			set_variable = {
				name = ve_religious_project_progress
				value = 0
			}
		}
		else_if = {
			limit = { var:ve_religious_project_progress >= 100 }
			set_variable = {
				name = ve_religious_project_progress
				value = 100
			}
			ve_finish_religious_project = yes
		}
	}
}
