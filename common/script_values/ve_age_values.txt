# This does not alter objective satisfaction or the first-completion reward.

		limit = { has_global_variable = ve_age_2_started }
		if = { limit = { has_modifier = extraction_idea_4_mdf } add = 1 }
		if = { limit = { has_modifier = agrarian_idea_4_mdf } add = 1 }

	else_if = {
		limit = { has_global_variable = ve_age_3_started }
		if = { limit = { has_modifier = offensive_idea_4_mdf } add = 1 }
		if = { limit = { has_modifier = defensive_idea_4_mdf } add = 1 }
		if = { limit = { has_modifier = naval_idea_4_mdf } add = 1 }
		if = { limit = { has_modifier = professional_army_idea_4_mdf } add = 1 }
		if = { limit = { has_modifier = mass_conscription_idea_4_mdf } add = 1 }
		if = { limit = { has_modifier = war_economy_idea_4_mdf } add = 1 }
	}
	min = 0
	max = 2
}

ve_era_reward_cost = {
	if = {
		limit = { var:ve_age_rewards_purchased <= 0 }
		value = 200
	}
	else_if = {
		limit = { var:ve_age_rewards_purchased == 1 }
		value = 300
	}
	else_if = {
		limit = { var:ve_age_rewards_purchased == 2 }
		value = 400
	}
	else_if = {
		limit = { var:ve_age_rewards_purchased == 3 }
		value = 500
	}
	else = {
		value = 600
	}
}
