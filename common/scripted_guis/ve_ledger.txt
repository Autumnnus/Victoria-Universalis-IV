# Doctrine Ledger bindings for gui/ve_panel_ledger.gui.
#
# All of these are executed with the viewing player's country as root; the
# rebuild's filters depend on that (see ve_ledger_triggers.txt).
# Effects here only touch the ledger's own presentation globals, never country
# state, so a ledger click can never change the game.

### OPEN AND REFRESH #########################################################

ve_ledger_refresh = {
	scope = country
	effect = { ve_ledger_rebuild = yes }
}

# True while the current page holds no rows: either nothing matches the filter
# or the ledger has not been built yet in this session.
ve_ledger_is_empty = {
	scope = country
	is_shown = {
		OR = {
			NOT = { has_global_variable = ve_ledger_total }
			global_var:ve_ledger_total == 0
		}
	}
}

### PAGING ###################################################################

ve_ledger_page_next = {
	scope = country
	is_valid = {
		has_global_variable = ve_ledger_offset
		has_global_variable = ve_ledger_total
		global_var:ve_ledger_total > 20
	}
	effect = { ve_ledger_next_page = yes }
}

ve_ledger_page_previous = {
	scope = country
	is_valid = {
		has_global_variable = ve_ledger_offset
		global_var:ve_ledger_offset >= 20
	}
	effect = { ve_ledger_previous_page = yes }
}

ve_ledger_page_first = {
	scope = country
	is_valid = {
		has_global_variable = ve_ledger_offset
		global_var:ve_ledger_offset >= 20
	}
	effect = { ve_ledger_first_page = yes }
}
